* Aparece nos relatórios futuros

* Alerta vermelho

* Pagamento confirmado com comprovante
* Entra no fluxo de caixa real
* Base para comparações

class EquipmentModel(db.Model):
    """Modelo genérico (ex: SM58, X32, Moving Head)"""
    id = db.Column(db.Integer, primary_key=True)
    company_id = db.Column(db.Integer, db.ForeignKey('company.id'))
    
    # Info modelo
    brand = db.Column(db.String(100))  # Shure
    model = db.Column(db.String(100))  # SM58
    type_id = db.Column(db.Integer, db.ForeignKey('equipment_type.id'))
    
    # FOTO COMPARTILHADA (única)
    foto_url = db.Column(db.String(500))  # /static/models/shure_sm58.jpg
    
    # Especificações
    specs = db.Column(db.Text)  # JSON com especificações
    valor_referencia = db.Column(db.Numeric(10, 2))
    
    # Relacionamento
    equipments = db.relationship('Equipment', backref='model', lazy='dynamic')


class Equipment(db.Model):
    """Unidade individual"""
    id = db.Column(db.Integer, primary_key=True)
    company_id = db.Column(db.Integer, db.ForeignKey('company.id'))
    
    code = db.Column(db.String(50), unique=True)  # SM58-001
    
    # REFERÊNCIA AO MODELO (sem foto própria!)
    model_id = db.Column(db.Integer, db.ForeignKey('equipment_model.id'))
    
    # Dados individuais
    serial_number = db.Column(db.String(100))
    data_compra = db.Column(db.Date)
    valor_compra = db.Column(db.Numeric(10, 2))
    status = db.Column(db.String(20))  # available, allocated, maintenance
    
    # QR/NFC/RFID (individual)
    qr_code = db.Column(db.String(200))
    nfc_tag = db.Column(db.String(100))
    rfid_tag = db.Column(db.String(100))
    
    # SEM FOTO AQUI! Usa model.foto_url
